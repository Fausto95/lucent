```ts title="notifications.lucent.ts"
import { PLATFORM } from "lucent:platform";
import { available } from "lucent:ios";
import { appContext } from "lucent:android";
import { UIApplication } from "lucent:ios/UIKit";
import { UNUserNotificationCenter } from "lucent:ios/UserNotifications";
import { NotificationChannel, NotificationManager } from "lucent:android/android.app";
import { Build_VERSION } from "lucent:android/android.os";
import { main } from "lucent:thread";

/** Sets the number on the app's icon. Android launchers show their own badges. */
export async function setBadge(count: number): Promise<void> {
  if (PLATFORM !== "ios") return;

  if (available("ios", 16)) {
    await UNUserNotificationCenter.current().setBadgeCount(BigInt(count));
  } else {
    await main(() => {
      UIApplication.shared.applicationIconBadgeNumber = BigInt(count);
    });
  }
}

/** Creates a notification channel, which Android 8 (API 26) added. */
export async function createChannel(id: string, name: string): Promise<void> {
  if (PLATFORM !== "android") return;
  if (Build_VERSION.SDK_INT < 26) return;

  const channel = new NotificationChannel(id, name, NotificationManager.IMPORTANCE_DEFAULT);
  appContext().getSystemService(NotificationManager)?.createNotificationChannel(channel);
}
```

An SDK declaration's doc comment gives the version that added it, such as `Since iOS 16.0.` or `Since API 26.`: [`lucent sdk show`](/docs/guides/find-an-sdk-class/) prints it. A class or member newer than [the oldest OS your app runs on](#the-oldest-os-your-app-runs-on) compiles only where your code checks the running OS first. Without the check, the build stops:

## What counts as a check

The compiler looks for the check around the call, in the source:

- [`available("ios", major, minor)`](/docs/api/lucent-ios/#available) from `lucent:ios` is Swift's `#available`, with `minor` 0 when left out.
- [`available("android", level)`](/docs/api/lucent-android/#available) from `lucent:android` is `Build.VERSION.SDK_INT >= level`. `Build_VERSION.SDK_INT` compared with a number, by `>=`, `>`, `<` or `<=` on either side, works too.
- A check guards the true side of an `if`, a `?:` or an `&&`. The opposite check, such as `!available("ios", 16)` or `Build_VERSION.SDK_INT < 26`, guards the `else`, the false side of a `?:` and what follows `||`.
- An `if` on the opposite check guards the statements after it, as in `createChannel()`. It has no `else`, and its branch ends in `return`, `throw`, `break` or `continue`.
- The version is a number literal: `Build_VERSION_CODES.O` or a variable isn't read as a check. A check for an older version than the API's doesn't count.
- Code written inside the check is covered, `main()` functions included. A function it calls needs its own check.

## The oldest OS your app runs on

On iOS, it's your app's deployment target: the app target's `IPHONEOS_DEPLOYMENT_TARGET` in the Xcode project under `ios/`, else the project's. An app deployed to iOS 16 calls an iOS 16 API without a check. Without an `ios/` project, as in an Expo app before `npx expo prebuild`, it's the oldest iOS on [Compatibility](/docs/api/compatibility/).

On Android, it's the API level on Compatibility, whatever your app's `minSdkVersion`. No roadmap task changes it.

A package's `ios.deploymentTarget` or `android.minSdk` in [lucent.json](/docs/packages/lucent-json/) raises the native package's, not these.
