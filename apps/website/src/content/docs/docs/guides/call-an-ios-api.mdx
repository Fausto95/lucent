```ts title="device.lucent.ts"
import { PLATFORM } from "lucent:platform";
import { errorCode } from "lucent:core";
import { FileManager, NSHomeDirectory } from "lucent:ios/Foundation";
import { UIDevice } from "lucent:ios/UIKit";
import { UNAuthorizationStatus, UNUserNotificationCenter } from "lucent:ios/UserNotifications";
import { main } from "lucent:thread";

/** The device's model, and the ID it gives this app's vendor. */
export async function device(): Promise<{ model: string; vendorId: string | null }> {
  if (PLATFORM !== "ios") throw new Error("iOS only");

  // UIDevice is main-thread only: main() runs the function there.
  return main(() => ({
    model: UIDevice.current.model,
    // NSUUID | null: nil after a restart, until the device is unlocked.
    vendorId: UIDevice.current.identifierForVendor?.uuidString ?? null,
  }));
}

/** Whether the person allows notifications. */
export async function notificationsAllowed(): Promise<boolean> {
  if (PLATFORM !== "ios") throw new Error("iOS only");

  // getNotificationSettings(completionHandler:) is also a promise.
  const settings = await UNUserNotificationCenter.current().notificationSettings();
  return settings.authorizationStatus === UNAuthorizationStatus.authorized;
}

/** The names in a folder of the app's Documents, or none when it doesn't exist. */
export async function list(folder: string): Promise<string[]> {
  if (PLATFORM !== "ios") throw new Error("iOS only");

  const path = `${NSHomeDirectory()}/Documents/${folder}`;
  try {
    return FileManager.default.contentsOfDirectory(path);
  } catch (e) {
    // An NSError's code is "<domain>:<code>": NSFileReadNoSuchFileError.
    if (errorCode(e as Error) === "NSCocoaErrorDomain:260") return [];
    throw e;
  }
}
```

<TabItem label="JS usage">

```tsx title="App.tsx"
import { device, list, notificationsAllowed } from "./src/device.lucent";

export async function deviceReport(): Promise<string> {
  const { model, vendorId } = await device(); // "iPhone", "6B29FC40-…"
  const allowed = await notificationsAllowed(); // false until the person allows them
  const names = await list("missing"); // []
  return `${model} ${vendorId ?? "locked"} ${allowed} ${names.length}`;
}
```

Each framework of the iOS SDK imports as `lucent:ios/<Framework>`, and so does each of the app's pods. Classes, properties and methods keep their Swift names, with the argument labels dropped: `contentsOfDirectory(atPath:)` is `contentsOfDirectory(path)`.

After the [guard clause](/docs/api/language/platform-code/#guard-clauses) on `PLATFORM`, the rest of each function is iOS code. [Call iOS and Android from one module](/docs/guides/platform-code/) adds the Android side.

Rebuild the app after you add or change the module: [Get a module change into the running app](/docs/guides/apply-a-change/) says what each build asks for.

## What the module shows

- **Main thread.** `UIDevice` is main-thread only, so `device()` reads it inside `main()`. In code the main thread doesn't run, it fails to compile with `LUCENT3006` ([Threads and versions](/docs/api/ios-sdk/#threads-and-versions)). [Call a main-thread-only API](/docs/guides/run-on-the-main-thread/) shows the pattern.
- **Optionals.** A Swift optional is `T | null`, so `?.` and `??` handle `identifierForVendor`, as [Types](/docs/api/ios-sdk/#types) says.
- **Completion handlers.** A method with a completion handler that Swift also imports as `async` has a second form. It returns a promise, under the `async` form's name: see [Blocks and completion handlers](/docs/api/ios-sdk/#blocks-and-completion-handlers).
- **Errors.** A method that throws in Swift throws an `Error`, and `errorCode()` from `lucent:core` reads its code, `"<domain>:<code>"` ([Errors](/docs/api/ios-sdk/#errors)). [Throw errors JavaScript can catch](/docs/guides/throw-and-handle-errors/) shows the JavaScript side.
- **Plain results.** SDK objects [can't cross](/docs/api/language/boundary/#what-cant-cross) to JavaScript, so each function returns plain values: an object of strings, a boolean, an array of strings.

An API newer than the oldest iOS Lucent supports compiles only under a [version check](/docs/guides/check-the-os-version/).

[iOS SDK bindings](/docs/api/ios-sdk/) gives every rule: names, types, out parameters, blocks and errors. Swift-only APIs, such as StoreKit 2 and CryptoKit, are called the same way: see [Call a Swift-only API](/docs/guides/call-a-swift-only-api/).

:::note
The first time your code imports a framework, Lucent reads its declarations from Xcode and caches them. `lucent sdk prefetch` does it ahead of time.
:::
