module native-union

export struct Result { kind: string, value: optional<float64>, message: optional<string> }

export fn evaluate(value: float64) -> struct Result
  if (lt value 0)
    return (struct Result (kind "error") (value null) (message "Negative"))
  return (struct Result (kind "ok") (value value) (message null))

export fn read(result: struct Result) -> float64
  if (eq (field result kind) "ok")
    return (unwrap (field result value))
  return 0
