module throw

export fn divide(a: float64, b: float64) -> float64
  if (eq b 0)
    throw "DIVIDE_BY_ZERO" "Cannot divide by zero"
  return (div a b)

export fn fail() -> void
  throw "ALWAYS"

export fn failWithMetadata(path: string) -> void
  throw "MISSING" "File\n不存在 🌍" metadata { path: path, attempt: 1, retry: false, detail: null }
