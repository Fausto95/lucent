module struct-roundtrip

export struct User { id: string, age: int32, nickname: optional<string>, tags: array<string> }

export fn birthday(user: struct User) -> struct User
  return (struct User (id (field user id)) (age (add (field user age) 1)) (nickname (field user nickname)) (tags (field user tags)))

export fn describe(user: struct User) -> string
  let %nickname: optional<string> = (field user nickname)
  if (eq %nickname null)
    return (concat (field user id) " (" (str (field user age)) ")")
  return (concat (unwrap %nickname) " aka " (field user id))
