module ternary-call

fn pick(value: float64) -> float64
  return (mul value 2)

export fn choose(flag: bool) -> float64
  let %chosen: float64 = (if flag (call pick 1) (call pick 2))
  return %chosen

export fn nested(flag: bool, other: bool) -> float64
  return (if flag (call pick 3) (if other (call pick 4) 5))
